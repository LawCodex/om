# مرسوم سلطاني رقم ٥٤ / ٨٦ بمنح الجنسية العمانية

نحن قابوس بن سعيد سلطان عمان

بعد الاطلاع على المرسوم السلطاني رقم ٢٦ / ٧٥ بإصدار قانون تنظيم الجهاز الإداري للدولة وتعديلاته،

وعلى المرسوم السلطاني رقم ٣ / ٨٣ بقانون تنظيم الجنسية العمانية وتعديلاته،

وبناء على ما تقتضيه المصلحة العامة.

رسمنا بما هو آت

## مادة (١)

تمنح الجنسية العمانية إلى كل من:

١- نصرة كامل أحمد خلف الله

٢- نادية عطية صادق

٣- فتحية محمد عثمان

٤- فائزة زكي عبدالعظيم

٥- نبيلة عبدالمجيد محمد

٦- فاطمة ابراهيم عبدالنبي شحاته

٧- نادية صلاح مهران

٨- وسيم فاطمة أحمد ناصر

٩- فاطمة حسن محمد

١٠- فخرية عبدالعزيز ابراهيم خلف

١١- وضحة محمد جمعة

١٢- آمال محمد سلامة عيد

١٣- اصلاح عربي محمد

١٤- عائشة عمران جمعان

١٥- نادية علي شعراوي

١٦- شاهين نظام الدين

١٧- حسيبة أنور عبدالرحمن

١٨- نعيمة حسن ابراهيم

١٩- ايمان علي أحمد الطاطوني

٢٠- انور جيهان حاجي بادشاة كل

٢١- فريدة محمد كاظم

٢٢- رقية بيجم عبود سالم بانعمان

٢٣- ماجدة بهجت عبد الظاهر محمد

٢٤- فوزية بنت عبدالرزاق بن هاشم

٢٥- نفيسة اسماعيل محمود السيد

٢٦- حميدة موسى حسين

٢٧- خاتون بنت خدا بخش الادين

٢٨- أحمدي عقيل عبدالله باعياش

٢٩- عائشة علي محمد البكري

٣٠- آمنة محمد عبدالواحد

٣١- سلمى بي بنت الله بخش بن دستجير شيخ

٣٢- حسينة فقير محمد خدا بخش

٣٣- فاطمة عبدالله علي

٣٤- زينب محمد نواز خان

٣٥- زخصانة بيكم بشير أحمد

٣٦- فاطمة خواجا بيكم محبوب خان

٣٧- كوهر عبدالستار خان محمد علي خان

٣٨- اختر بيجم قادر محمد

٣٩- محموده بيكم حبيب الدين عبدالرحمن

٤٠- نورجهان اسلم خان

٤١- زبيدة بني يانو ابراهيم

٤٢- ساردار بانو محمد ابراهيم خان

٤٣- راميساني زيد اسماعيل باتان

٤٤- شعيم سلطانه محمد اسكندر

٤٥- عائشة بي برسبا بارابا

٤٦- خيرية عبده ابراهيم

٤٧- جميلة عبدالسلام ابراهيم سالم

٤٨- نداء عبدالله شحاته

٤٩- ثناء مصطفى حسن

٥٠- سمارة حافظ حميده

٥١- نجاة أحمد محمد خميس

٥٢- نجمة فضل محمد

٥٣- ماهاتون حسن نجمان البلوشي

٥٤- نعيمة سالم ناصر العمري

٥٥- هدى شاناز حسين

٥٦- عائشة حسن أحمد

٥٧- سلطانه بي منزور خان

٥٨- رحمة النساء سخاوت حسين

٥٩- نبيهه محمود عبدالفتاح

٦٠- سمير جيمي فانس

## مادة (٢)

ينشر هذا المرسوم في الجريدة الرسمية ويعمل به اعتبارا من تاريخ نشره.

**صدر في: ٢ ذو القعدة سنة ١٤٠٦هـ**

**الموافق: ١٠ يوليو سنة ١٩٨٦م**

**قابوس بن سعيد**

**سلطان عمان**

نشر هذا المرسوم في عدد الجريدة الرسمية رقم (٣٣٩) الصادر في ١٥ / ٧ / ١٩٨٦م.
